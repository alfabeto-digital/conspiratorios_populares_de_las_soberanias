{{- /* Render-hook de enlaces para info.. - http(s)/mailto/anclas -> se dejan igual. - enlace que escapa de la bóveda (../) -> navegador de código (GitHub). - *.excalidraw.md -> su PNG (sirve la visualización). - *.md interno -> la página Hugo correspondiente. - otro archivo interno -> ruta estática absoluta. */ -}} {{- $dest := .Destination -}} {{- $text := .Text -}} {{- $title := .Title -}} {{- $u := urls.Parse $dest -}} {{- if or $u.IsAbs (strings.HasPrefix $dest "#") (strings.HasPrefix $dest "mailto:") -}} {{ $text | safeHTML }} {{- else -}} {{- $parts := split $dest "#" -}} {{- $path := index $parts 0 -}} {{- $frag := "" -}} {{- if gt (len $parts) 1 -}}{{- $frag = printf "#%s" (index $parts 1) -}}{{- end -}} {{- $resolved := "" -}} {{- if strings.HasPrefix $path "/" -}} {{- $resolved = strings.TrimPrefix "/" $path -}} {{- else -}} {{- $resolved = path.Join (path.Dir .Page.File.Path) $path -}} {{- end -}} {{- if strings.HasPrefix $resolved ".." -}} {{- $repoPath := strings.TrimPrefix "../" $resolved -}} {{ $text | safeHTML }} {{- else if strings.HasPrefix $resolved "graphify/" -}} {{- if eq $resolved "graphify/graph.html" -}} {{/* el grafo interactivo se sirve estático */}} {{ $text | safeHTML }} {{- else -}} {{/* reportes, cache, scripts de graphify -> navegador de código */}} {{ $text | safeHTML }} {{- end -}} {{- else if strings.HasSuffix $path ".excalidraw.md" -}} {{- $png := replaceRE "(-slideshow)?\\.excalidraw\\.md$" ".png" $resolved -}} {{ $text | safeHTML }} {{- else if strings.HasSuffix $path ".md" -}} {{- $ref := printf "/%s" (strings.TrimSuffix ".md" $resolved) -}} {{- $pg := site.GetPage $ref -}} {{- if not $pg -}}{{- $pg = site.GetPage (path.Dir $ref) -}}{{- end -}} {{- with $pg -}} {{ $text | safeHTML }} {{- else -}} {{ $text | safeHTML }} {{- end -}} {{- else -}} {{ $text | safeHTML }} {{- end -}} {{- end -}}
