{{- /* Render-hook de imágenes para info.. - http(s) -> se deja igual. - ruta absoluta (/...) -> estático (viene del preprocesado de embeds ![[...]]). - ruta relativa -> se resuelve a ruta estática absoluta. - ruta que escapa (../) -> navegador de código. */ -}} {{- $dest := .Destination -}} {{- $u := urls.Parse $dest -}} {{- if $u.IsAbs -}} {{- else if strings.HasPrefix $dest "/" -}} {{- else -}} {{- $pageDir := path.Dir .Page.File.Path -}} {{- $resolved := path.Join $pageDir $dest -}} {{- if strings.HasPrefix $resolved ".." -}} {{- else -}} {{- end -}} {{- end -}}
